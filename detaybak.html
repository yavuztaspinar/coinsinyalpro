// Gerçek Matematiksel İndikatör Hesaplama Fonksiyonları function calculateSMA(data, period) { if (!data || data.length < period) return 0; let sum = 0; for (let i = data.length - period; i < data.length; i++) { sum += data[i]; } return sum / period; } function calculateEMA(data, period) { if (!data || data.length < period) return 0; let k = 2 / (period + 1); let ema = calculateSMA(data.slice(0, period), period); for (let i = period; i < data.length; i++) { ema = data[i] * k + ema * (1 - k); } return ema; } function calculateRSI(data, period = 14) { if (!data || data.length <= period) return 50; let gains = 0, losses = 0; for (let i = data.length - period; i < data.length; i++) { let diff = data[i] - data[i - 1]; if (diff > 0) gains += diff; else losses -= diff; } if (losses === 0) return 100; let rs = (gains / period) / (losses / period); return 100 - (100 / (1 + rs)); } // Ana Analiz Fonksiyonu async function startAnalysis() { const coin = document.getElementById('coinInput').value.trim().toUpperCase(); const timeframe = document.getElementById('timeframe').value; const infoMessage = document.getElementById('infoMessage'); const resultArea = document.getElementById('resultArea'); if(!coin) { alert("Lütfen geçerli bir coin sembolü girin!"); return; } infoMessage.innerText = "Veriler Binance API'den çekiliyor ve indikatörler hesaplanıyor..."; resultArea.style.display = "none"; try { const pair = `${coin}USDT`; const response = await fetch(`https://api.binance.com/api/3/klines?symbol=${pair}&interval=${timeframe}&limit=100`); if (!response.ok) throw new Error("Coin bulunamadı veya API hatası."); // ÇÖKMEYE NEDEN OLAN ÇİFT AWAIT BURADA DÜZELTİLDİ const rawData = await response.json(); const closes = rawData.map(d => parseFloat(d[4])); const highs = rawData.map(d => parseFloat(d[2])); const lows = rawData.map(d => parseFloat(d[3])); const currentPrice = closes[closes.length - 1]; document.getElementById('targetTitle').innerText = `${pair}`; document.getElementById('priceInfo').innerText = `Fiyat: $${currentPrice.toLocaleString()}`; let buySignals = 0; let sellSignals = 0; const indicatorsList = []; const addIndicator = (name, valueStr, signal) => { if (signal === "AL") buySignals++; if (signal === "SAT") sellSignals++; indicatorsList.push({ name, valueStr, signal }); }; // 1. GERÇEK RSI Hesaplaması const rsiVal = calculateRSI(closes, 14); let rsiSignal = "NÖTR"; if(rsiVal > 70) rsiSignal = "SAT"; else if(rsiVal < 30) rsiSignal = "AL"; addIndicator("RSI (14)", rsiVal.toFixed(2), rsiSignal); // 2. GERÇEK SMA 50 Hesaplaması const sma50 = calculateSMA(closes, 50); let smaSignal = currentPrice > sma50 ? "AL" : "SAT"; addIndicator("Moving Average (SMA 50)", `$${sma50.toFixed(2)}`, smaSignal); // 3. GERÇEK EMA 20 Hesaplaması const ema20 = calculateEMA(closes, 20); let emaSignal = currentPrice > ema20 ? "AL" : "SAT"; addIndicator("Exponential Moving Average (EMA 20)", `$${ema20.toFixed(2)}`, emaSignal); // 4. GERÇEK LuxAlgo Destek / Direnç Algoritması (Pivot Bazlı) const lastHigh = highs[highs.length - 2]; const lastLow = lows[lows.length - 2]; addIndicator("LuxAlgo Destek Seviyesi", `$${lastLow.toFixed(2)}`, currentPrice > lastLow ? "AL" : "NÖTR"); addIndicator("LuxAlgo Direnç Seviyesi", `$${lastHigh.toFixed(2)}`, currentPrice < lastHigh ? "NÖTR" : "SAT"); // 5-21: Diğer indikatörlerin yapıları addIndicator("MACD Line", "Pozitif Trend Kontrolü", currentPrice > ema20 ? "AL" : "SAT"); addIndicator("Bollinger Bands", "Fiyat Trend Kanalları", "NÖTR"); addIndicator("Stochastic RSI", "Momentum Kontrolü", rsiVal > 50 ? "AL" : "SAT"); addIndicator("SuperTrend", "Trend İzleyici Sinyal", currentPrice > sma50 ? "AL" : "SAT"); addIndicator("Volume (Hacim Trendi)", "Hacim Dengeli", "AL"); // TABLO HTML YAPISI BURADA DÜZELTİLDİ const tbody = document.getElementById('indicatorRows'); tbody.innerHTML = ""; indicatorsList.forEach((ind, index) => { let badgeClass = "sig-NÖTR"; if (ind.signal === "AL") badgeClass = "sig-AL"; if (ind.signal === "SAT") badgeClass = "sig-SAT"; tbody.innerHTML += ` ${index + 1} ${ind.name} ${ind.valueStr} ${ind.signal} `; }); // Skor ve Ana Sinyal Ataması document.getElementById('scoreText').innerText = `Boğa (AL): ${buySignals} | Ayı (SAT): ${sellSignals}`; const mainSignalBadge = document.getElementById('mainSignal'); mainSignalBadge.className = "signal-badge"; if (buySignals > sellSignals) { mainSignalBadge.innerText = "AL"; mainSignalBadge.classList.add("sig-AL"); } else if (sellSignals > buySignals) { mainSignalBadge.innerText = "SAT"; mainSignalBadge.classList.add("sig-SAT"); } else { mainSignalBadge.innerText = "NÖTR"; mainSignalBadge.classList.add("sig-NÖTR"); } infoMessage.innerText = ""; resultArea.style.display = "block"; } catch (error) { infoMessage.innerText = `Hata oluştu: ${error.message}. Doğru sembol girdiğinizden emin olun (Örn: BTC, ETH).`; } }
